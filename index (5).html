// ====================================================================
// BACKEND GOOGLE APPS SCRIPT - LP AL BAITUL AMIEN JEMBER
// ====================================================================

const SHEET_NAME = 'DATABASE_LAPORAN';

function doGet(e) {
  return HtmlService.createTemplateFromFile('index')
    .evaluate()
    .setTitle('Hotline Pengaduan - Lembaga Pendidikan Al Baitul Amien Jember')
    .addMetaTag('viewport', 'width=device-width, initial-scale=1');
}

// SIMPAN LAPORAN BARU
function submitLaporanGAS(data) {
  try {
    const sheet = SpreadsheetApp.getActiveSpreadsheet().getSheetByName(SHEET_NAME);
    const now = new Date();
    const dateCode = Utilities.formatDate(now, 'GMT+7', 'yyyyMMdd');
    const randomNum = Math.floor(1000 + Math.random() * 9000);
    const idLaporan = 'ABA-' + dateCode + '-' + randomNum;
    const tglStr = Utilities.formatDate(now, 'GMT+7', 'yyyy-MM-dd HH:mm');

    sheet.appendRow([
      idLaporan,
      tglStr,
      data.nama,
      data.kontak,
      data.peran,
      data.unit,
      data.kategori,
      data.judul,
      data.detail,
      'Menunggu',
      '',
      data.lampiran || ''
    ]);

    return { success: true, id: idLaporan };
  } catch (err) {
    return { success: false, message: err.toString() };
  }
}

// AMBIL SEMUA DATA LAPORAN (UNTUK ADMIN)
function getLaporanGAS() {
  const sheet = SpreadsheetApp.getActiveSpreadsheet().getSheetByName(SHEET_NAME);
  if (!sheet) return [];
  const data = sheet.getDataRange().getValues();
  if (data.length <= 1) return [];
  data.shift(); // Hapus Header
  return data;
}

// UPDATE STATUS & TANGGAPAN PETUGAS
function updateLaporanGAS(idLaporan, status, catatan) {
  const sheet = SpreadsheetApp.getActiveSpreadsheet().getSheetByName(SHEET_NAME);
  const data = sheet.getDataRange().getValues();
  
  for (let i = 1; i < data.length; i++) {
    if (data[i][0] === idLaporan) {
      sheet.getRange(i + 1, 10).setValue(status);  // Kolom Status (J)
      sheet.getRange(i + 1, 11).setValue(catatan); // Kolom Catatan (K)
      return { success: true };
    }
  }
  return { success: false, message: 'ID Laporan tidak ditemukan.' };
}

// AUTO SETUP TAB SPREADSHEET JIKA BELUM ADA
function setupSpreadsheet() {
  const ss = SpreadsheetApp.getActiveSpreadsheet();
  let sheet = ss.getSheetByName(SHEET_NAME);
  if (!sheet) {
    sheet = ss.insertSheet(SHEET_NAME);
    sheet.appendRow([
      'ID Laporan', 'Tanggal', 'Nama Pelapor', 'No WA', 'Peran', 
      'Unit', 'Kategori', 'Judul', 'Detail Kronologi', 'Status', 'Catatan Admin', 'Lampiran URL'
    ]);
    sheet.getRange("1:1").setFontWeight("bold").setBackground("#065f46").setFontColor("#ffffff");
  }
}
